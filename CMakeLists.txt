cmake_minimum_required(VERSION 3.16)
project(economic_simulation LANGUAGES CXX)
add_executable(economy src/main.cpp)
target_compile_features(economy PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(economy PRIVATE /W4)
else()
  target_compile_options(economy PRIVATE -Wall -Wextra -Wpedantic)
endif()
